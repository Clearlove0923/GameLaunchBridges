cmake_minimum_required(VERSION 3.24)
project(ArknightsEndfieldLaunchBridge VERSION 1.0.0 LANGUAGES CXX RC)
function(configure_native target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_options(${target} PRIVATE /MANIFEST:NO)
  target_link_libraries(${target} PRIVATE shell32 ole32 dbghelp version advapi32)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endfunction()
add_executable(ArknightsEndfieldLaunchBridge WIN32 src/main.cpp src/app.rc)
configure_native(ArknightsEndfieldLaunchBridge)
add_executable(ArknightsEndfieldLaunchBridgeTests tests/main.cpp src/app.rc)
configure_native(ArknightsEndfieldLaunchBridgeTests)
add_executable(EndfieldFixture WIN32 tests/fixture.cpp src/app.rc)
configure_native(EndfieldFixture)
set_target_properties(EndfieldFixture PROPERTIES OUTPUT_NAME Endfield RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixture")
